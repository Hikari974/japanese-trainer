# US-006.4: UI Sélection Niveau avec États Locked/Unlocked

**Epic:** [Epic-006](./epic-006.md) - Système de Progression et Déblocage Séquentiel
**Priorité:** P1 (High)
**Effort Estimé:** L (1-1.5 jours)

## User Story

> **En tant qu'** utilisateur naviguant dans l'application
> **Je veux** voir clairement quels niveaux JLPT sont débloqués/verrouillés
> **Afin de** comprendre ma progression et savoir sur quels niveaux je peux m'entraîner

## Critères d'Acceptation

### Fonctionnel
- [ ] **AC1:** Niveaux locked affichés avec icône cadenas + opacité réduite
- [ ] **AC2:** Niveaux locked non cliquables (disabled)
- [ ] **AC3:** Tooltip/texte explicatif sur niveaux locked ("Complétez X pour débloquer")
- [ ] **AC4:** Niveau current visuellement distinct (border/background accentué)
- [ ] **AC5:** Niveaux completed affichés avec badge checkmark vert

### UX
- [ ] **AC6:** Transition fluide 60fps lors changement état (locked → unlocked)
- [ ] **AC7:** Touch target >= 44px même pour niveaux locked
- [ ] **AC8:** Feedback tactile (vibration légère) sur tentative tap niveau locked
- [ ] **AC9:** Progression affichée sur chaque niveau (X/Y mots maîtrisés)
- [ ] **AC10:** Auto-scroll vers premier niveau unlocked au mount

### Technique
- [ ] **AC11:** États récupérés via `StatsManager.getUnlockedLevels()` et `getLevelProgress()`
- [ ] **AC12:** Utilisation hooks React (`useState`, `useEffect`) pour refresh auto
- [ ] **AC13:** Animations Tamagui (pas de libs externes)
- [ ] **AC14:** Accessible (labels ARIA pour screen readers)
- [ ] **AC15:** Pas de régression sur sélection niveau existante

### Implémentation React
```typescript
// app/components/LevelSelect.tsx
import React, { useEffect, useState } from 'react';
import { YStack, XStack, Button, Text, Progress } from 'tamagui';
import { Lock, Check } from '@tamagui/lucide-icons';
import StatsManager from '../core/training/StatsManager';
import { JLPTLevel, JLPT_LEVEL_ORDER, LevelProgress } from '../core/training/types';

interface LevelSelectProps {
  selectedLevel: JLPTLevel;
  onLevelSelect: (level: JLPTLevel) => void;
}

export default function LevelSelect({ selectedLevel, onLevelSelect }: LevelSelectProps) {
  const [unlockedLevels, setUnlockedLevels] = useState<JLPTLevel[]>([]);
  const [levelsProgress, setLevelsProgress] = useState<Map<JLPTLevel, LevelProgress>>(new Map());
  const [loading, setLoading] = useState(true);

  // Charger états déblocage + progression au mount
  useEffect(() => {
    loadLevelStates();
  }, []);

  async function loadLevelStates() {
    setLoading(true);
    const manager = StatsManager.getInstance();

    const unlocked = await manager.getUnlockedLevels();
    const progress = await manager.getAllLevelsProgress();

    setUnlockedLevels(unlocked);
    setLevelsProgress(progress);
    setLoading(false);
  }

  // Déterminer état visuel d'un niveau
  function getLevelState(level: JLPTLevel): 'locked' | 'unlocked' | 'current' | 'completed' {
    if (level === selectedLevel) return 'current';

    const progress = levelsProgress.get(level);
    if (progress?.isComplete) return 'completed';

    if (!unlockedLevels.includes(level)) return 'locked';

    return 'unlocked';
  }

  // Handler tap niveau
  function handleLevelTap(level: JLPTLevel) {
    const state = getLevelState(level);

    if (state === 'locked') {
      // Feedback tactile
      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning);
      return; // Pas de sélection
    }

    Haptics.selectionAsync();
    onLevelSelect(level);
  }

  if (loading) {
    return <Text>Chargement...</Text>;
  }

  return (
    <YStack gap="$3" padding="$4">
      {JLPT_LEVEL_ORDER.map(level => {
        const state = getLevelState(level);
        const progress = levelsProgress.get(level);
        const isLocked = state === 'locked';
        const isCurrent = state === 'current';
        const isCompleted = state === 'completed';

        return (
          <LevelButton
            key={level}
            level={level}
            state={state}
            progress={progress}
            onPress={() => handleLevelTap(level)}
          />
        );
      })}
    </YStack>
  );
}

// Composant bouton niveau (isolé pour perfs)
interface LevelButtonProps {
  level: JLPTLevel;
  state: 'locked' | 'unlocked' | 'current' | 'completed';
  progress?: LevelProgress;
  onPress: () => void;
}

function LevelButton({ level, state, progress, onPress }: LevelButtonProps) {
  const isLocked = state === 'locked';
  const isCurrent = state === 'current';
  const isCompleted = state === 'completed';

  return (
    <Button
      size="$6"
      disabled={isLocked}
      onPress={onPress}
      opacity={isLocked ? 0.4 : 1}
      borderWidth={isCurrent ? 2 : 1}
      borderColor={isCurrent ? '$blue10' : '$gray6'}
      backgroundColor={isCurrent ? '$blue3' : '$background'}
      pressStyle={{ scale: isLocked ? 1 : 0.97 }}
      animation="quick"
    >
      <XStack gap="$3" alignItems="center" width="100%">
        {/* Icône état */}
        {isLocked && <Lock size={20} color="$gray10" />}
        {isCompleted && <Check size={20} color="$green10" />}

        {/* Nom niveau */}
        <YStack flex={1}>
          <Text fontSize="$6" fontWeight="bold">
            {level.toUpperCase()}
          </Text>

          {/* Progression ou message locked */}
          {isLocked ? (
            <Text fontSize="$2" color="$gray10">
              Complétez {getPreviousLevelName(level)} pour débloquer
            </Text>
          ) : progress && (
            <>
              <Text fontSize="$3" color="$gray11">
                {progress.masteredWords}/{progress.totalWords} mots ({progress.percentage.toFixed(1)}%)
              </Text>
              <Progress
                value={progress.percentage}
                max={100}
                size="$2"
                marginTop="$2"
              >
                <Progress.Indicator animation="quick" backgroundColor="$blue10" />
              </Progress>
            </>
          )}
        </YStack>
      </XStack>
    </Button>
  );
}

// Helper: Nom niveau précédent pour message locked
function getPreviousLevelName(level: JLPTLevel): string {
  const index = JLPT_LEVEL_ORDER.indexOf(level);
  if (index <= 0) return '';
  return JLPT_LEVEL_ORDER[index - 1].toUpperCase();
}
```

## Cas de Test

### Tests Composant
```typescript
// __tests__/LevelSelect.test.tsx
import { render, fireEvent, waitFor } from '@testing-library/react-native';
import LevelSelect from '../LevelSelect';
import StatsManager from '../core/training/StatsManager';

jest.mock('../core/training/StatsManager');

describe('LevelSelect Component', () => {
  let mockStatsManager: jest.Mocked<StatsManager>;

  beforeEach(() => {
    mockStatsManager = {
      getUnlockedLevels: jest.fn().mockResolvedValue(['kana', 'n5']),
      getAllLevelsProgress: jest.fn().mockResolvedValue(new Map([
        ['kana', { totalWords: 92, masteredWords: 92, percentage: 100, isComplete: true }],
        ['n5', { totalWords: 120, masteredWords: 78, percentage: 65, isComplete: false }],
        ['n4', { totalWords: 150, masteredWords: 0, percentage: 0, isComplete: false }]
      ]))
    } as any;

    StatsManager.getInstance = jest.fn().mockReturnValue(mockStatsManager);
  });

  it('affiche tous les 6 niveaux JLPT', async () => {
    const { getByText } = render(
      <LevelSelect selectedLevel="kana" onLevelSelect={jest.fn()} />
    );

    await waitFor(() => {
      expect(getByText('KANA')).toBeTruthy();
      expect(getByText('N5')).toBeTruthy();
      expect(getByText('N4')).toBeTruthy();
      expect(getByText('N3')).toBeTruthy();
      expect(getByText('N2')).toBeTruthy();
      expect(getByText('N1')).toBeTruthy();
    });
  });

  it('affiche icône cadenas sur niveaux locked', async () => {
    const { getByTestId } = render(
      <LevelSelect selectedLevel="kana" onLevelSelect={jest.fn()} />
    );

    await waitFor(() => {
      expect(getByTestId('lock-icon-n4')).toBeTruthy();
      expect(getByTestId('lock-icon-n3')).toBeTruthy();
    });
  });

  it('affiche checkmark sur niveaux completed', async () => {
    const { getByTestId } = render(
      <LevelSelect selectedLevel="n5" onLevelSelect={jest.fn()} />
    );

    await waitFor(() => {
      expect(getByTestId('check-icon-kana')).toBeTruthy();
    });
  });

  it('affiche progression sur niveaux unlocked', async () => {
    const { getByText } = render(
      <LevelSelect selectedLevel="kana" onLevelSelect={jest.fn()} />
    );

    await waitFor(() => {
      expect(getByText('78/120 mots (65.0%)')).toBeTruthy();
    });
  });

  it('ne permet pas sélection niveau locked', async () => {
    const onLevelSelect = jest.fn();
    const { getByText } = render(
      <LevelSelect selectedLevel="kana" onLevelSelect={onLevelSelect} />
    );

    await waitFor(() => {
      const n4Button = getByText('N4').parent;
      fireEvent.press(n4Button);
    });

    expect(onLevelSelect).not.toHaveBeenCalled();
  });

  it('permet sélection niveau unlocked', async () => {
    const onLevelSelect = jest.fn();
    const { getByText } = render(
      <LevelSelect selectedLevel="kana" onLevelSelect={onLevelSelect} />
    );

    await waitFor(() => {
      const n5Button = getByText('N5').parent;
      fireEvent.press(n5Button);
    });

    expect(onLevelSelect).toHaveBeenCalledWith('n5');
  });

  it('applique style distinct au niveau current', async () => {
    const { getByText } = render(
      <LevelSelect selectedLevel="n5" onLevelSelect={jest.fn()} />
    );

    await waitFor(() => {
      const n5Button = getByText('N5').parent;
      expect(n5Button.props.style).toMatchObject({
        borderWidth: 2,
        borderColor: expect.stringContaining('blue')
      });
    });
  });
});
```
