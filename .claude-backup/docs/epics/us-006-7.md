### Implémentation

#### Composant Section Progression
```typescript
// app/components/LevelProgressSection.tsx
import React, { useEffect, useState } from 'react';
import { YStack, XStack, Text, Card, Progress, Button } from 'tamagui';
import { Lock, Check, Activity } from '@tamagui/lucide-icons';
import { useRouter } from 'expo-router';
import StatsManager from '@/core/training/StatsManager';
import { JLPTLevel, JLPT_LEVEL_ORDER, LevelProgress } from '@/core/training/types';

export default function LevelProgressSection() {
  const router = useRouter();
  const [levelsProgress, setLevelsProgress] = useState<Map<JLPTLevel, LevelProgress>>(new Map());
  const [unlockedLevels, setUnlockedLevels] = useState<JLPTLevel[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadProgressData();
  }, []);

  async function loadProgressData() {
    setLoading(true);
    const manager = StatsManager.getInstance();

    const progress = await manager.getAllLevelsProgress();
    const unlocked = await manager.getUnlockedLevels();

    setLevelsProgress(progress);
    setUnlockedLevels(unlocked);
    setLoading(false);
  }

  // Calculer stats globales
  const totalWords = Array.from(levelsProgress.values()).reduce(
    (sum, p) => sum + p.totalWords,
    0
  );
  const totalMastered = Array.from(levelsProgress.values()).reduce(
    (sum, p) => sum + p.masteredWords,
    0
  );
  const globalPercentage = totalWords > 0 ? (totalMastered / totalWords) * 100 : 0;
  const currentLevel = unlockedLevels[unlockedLevels.length - 1] || 'kana';

  function handleLevelTap(level: JLPTLevel) {
    const isLocked = !unlockedLevels.includes(level);
    if (isLocked) return;

    router.push(`/level-progress/${level}`);
  }

  if (loading) {
    return <Text>Chargement progression...</Text>;
  }

  return (
    <YStack gap="$4">
      {/* Header Stats Globales */}
      <Card padding="$4" backgroundColor="$blue2">
        <Text fontSize="$6" fontWeight="bold" marginBottom="$2">
          📊 Progression JLPT
        </Text>

        <XStack gap="$4" marginBottom="$3">
          <YStack flex={1}>
            <Text fontSize="$3" color="$gray11">Niveau actuel</Text>
            <Text fontSize="$6" fontWeight="bold" color="$blue11">
              {currentLevel.toUpperCase()}
            </Text>
          </YStack>

          <YStack flex={1}>
            <Text fontSize="$3" color="$gray11">Mots maîtrisés</Text>
            <Text fontSize="$6" fontWeight="bold">
              {totalMastered}/{totalWords}
            </Text>
          </YStack>
        </XStack>

        <Progress value={globalPercentage} max={100} size="$3">
          <Progress.Indicator animation="quick" backgroundColor="$blue10" />
        </Progress>

        <Text fontSize="$2" color="$gray10" marginTop="$2">
          {globalPercentage.toFixed(1)}% du parcours JLPT complété
        </Text>
      </Card>

      {/* Cartes Niveaux */}
      <YStack gap="$3">
        {JLPT_LEVEL_ORDER.map(level => {
          const progress = levelsProgress.get(level);
          const isLocked = !unlockedLevels.includes(level);
          const isCompleted = progress?.isComplete || false;

          return (
            <LevelCard
              key={level}
              level={level}
              progress={progress}
              isLocked={isLocked}
              isCompleted={isCompleted}
              onPress={() => handleLevelTap(level)}
            />
          );
        })}
      </YStack>
    </YStack>
  );
}

// Composant Carte Niveau
interface LevelCardProps {
  level: JLPTLevel;
  progress?: LevelProgress;
  isLocked: boolean;
  isCompleted: boolean;
  onPress: () => void;
}

function LevelCard({ level, progress, isLocked, isCompleted, onPress }: LevelCardProps) {
  const icon = isLocked ? <Lock size={20} /> : isCompleted ? <Check size={20} /> : <Activity size={20} />;
  const iconColor = isLocked ? '$gray10' : isCompleted ? '$green10' : '$blue10';
  const opacity = isLocked ? 0.5 : 1;

  return (
    <Card
      padding="$4"
      onPress={onPress}
      disabled={isLocked}
      opacity={opacity}
      pressStyle={{ scale: isLocked ? 1 : 0.98 }}
      animation="quick"
      borderWidth={1}
      borderColor={isCompleted ? '$green6' : isLocked ? '$gray5' : '$blue6'}
    >
      <XStack gap="$3" alignItems="center">
        {/* Icône état */}
        <YStack
          width={40}
          height={40}
          alignItems="center"
          justifyContent="center"
          backgroundColor={isCompleted ? '$green3' : isLocked ? '$gray3' : '$blue3'}
          borderRadius="$4"
        >
          {React.cloneElement(icon, { color: iconColor })}
        </YStack>

        {/* Info niveau */}
        <YStack flex={1} gap="$1">
          <XStack justifyContent="space-between" alignItems="center">
            <Text fontSize="$6" fontWeight="bold">
              {level.toUpperCase()}
            </Text>

            {progress && (
              <Text fontSize="$3" color="$gray11">
                {progress.masteredWords}/{progress.totalWords}
              </Text>
            )}
          </XStack>

          {/* Barre progression ou message locked */}
          {isLocked ? (
            <Text fontSize="$2" color="$gray10">
              Complétez {getPreviousLevel(level)} pour débloquer
            </Text>
          ) : progress && (
            <>
              <Progress value={progress.percentage} max={100} size="$2">
                <Progress.Indicator
                  backgroundColor={isCompleted ? '$green10' : '$blue10'}
                />
              </Progress>

              <Text fontSize="$2" color="$gray10">
                {progress.percentage.toFixed(1)}% maîtrisé
                {isCompleted && ' - Complété ✓'}
              </Text>
            </>
          )}
        </YStack>
      </XStack>
    </Card>
  );
}

function getPreviousLevel(level: JLPTLevel): string {
  const index = JLPT_LEVEL_ORDER.indexOf(level);
  if (index <= 0) return '';
  return JLPT_LEVEL_ORDER[index - 1].toUpperCase();
}
```

## Cas de Test

### Tests Composant
```typescript
// __tests__/LevelProgressSection.test.tsx
import { render, fireEvent, waitFor } from '@testing-library/react-native';
import LevelProgressSection from '../LevelProgressSection';
import StatsManager from '../core/training/StatsManager';

jest.mock('../core/training/StatsManager');

describe('LevelProgressSection', () => {
  beforeEach(() => {
    const mockProgress = new Map([
      ['kana', { totalWords: 92, masteredWords: 92, percentage: 100, isComplete: true }],
      ['n5', { totalWords: 800, masteredWords: 123, percentage: 15.4, isComplete: false }],
      ['n4', { totalWords: 1000, masteredWords: 0, percentage: 0, isComplete: false }]
    ]);

    StatsManager.getInstance = jest.fn().mockReturnValue({
      getAllLevelsProgress: jest.fn().mockResolvedValue(mockProgress),
      getUnlockedLevels: jest.fn().mockResolvedValue(['kana', 'n5'])
    });
  });

  it('affiche stats globales correctes', async () => {
    const { getByText } = render(<LevelProgressSection />);

    await waitFor(() => {
      expect(getByText('N5')).toBeTruthy(); // Niveau actuel
      expect(getByText('215/1892')).toBeTruthy(); // Total mots (92+123 / 92+800+1000)
    });
  });

  it('affiche 6 cartes niveaux', async () => {
    const { getByText } = render(<LevelProgressSection />);

    await waitFor(() => {
      expect(getByText('KANA')).toBeTruthy();
      expect(getByText('N5')).toBeTruthy();
      expect(getByText('N4')).toBeTruthy();
      expect(getByText('N3')).toBeTruthy();
      expect(getByText('N2')).toBeTruthy();
      expect(getByText('N1')).toBeTruthy();
    });
  });

  it('affiche icône checkmark sur niveau complété', async () => {
    const { getByTestId } = render(<LevelProgressSection />);

    await waitFor(() => {
      expect(getByTestId('check-icon-kana')).toBeTruthy();
    });
  });

  it('affiche icône cadenas sur niveau locked', async () => {
    const { getByTestId } = render(<LevelProgressSection />);

    await waitFor(() => {
      expect(getByTestId('lock-icon-n4')).toBeTruthy();
    });
  });

  it('navigue vers détail niveau au tap', async () => {
    const mockPush = jest.fn();
    jest.mock('expo-router', () => ({
      useRouter: () => ({ push: mockPush })
    }));

    const { getByText } = render(<LevelProgressSection />);

    await waitFor(() => {
      fireEvent.press(getByText('N5'));
    });

    expect(mockPush).toHaveBeenCalledWith('/level-progress/n5');
  });

  it('ne navigue pas si niveau locked', async () => {
    const mockPush = jest.fn();

    const { getByText } = render(<LevelProgressSection />);

    await waitFor(() => {
      fireEvent.press(getByText('N4'));
    });

    expect(mockPush).not.toHaveBeenCalled();
  });
});
```
