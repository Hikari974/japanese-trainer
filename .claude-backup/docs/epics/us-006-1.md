```typescript
function getLevelProgress(level: JLPTLevel): LevelProgress {
  // 1. Filtrer mots du niveau dans vocabulary.json
  const levelWords = vocabulary.filter(w => w.jlptLevel === level)

  // 2. Pour chaque mot, vérifier score >= 5
  const masteredWords = levelWords.filter(w =>
    (stats.wordScores[w.hiragana] || 0) >= 5
  )

  // 3. Retourner progression
  return {
    totalWords: levelWords.length,
    masteredWords: masteredWords.length,
    percentage: (masteredWords.length / levelWords.length) * 100,
    isComplete: masteredWords.length === levelWords.length
  }
}
```

### Méthode StatsManager
```typescript
// app/core/training/StatsManager.ts
import vocabulary from '../data/vocabulary.json';

class StatsManager {
  // ... méthodes existantes ...

  /**
   * Calcule la progression pour un niveau JLPT donné
   * @param level - Niveau JLPT ("kana", "n5", etc.)
   * @returns Objet LevelProgress avec détails progression
   */
  async getLevelProgress(level: JLPTLevel): Promise<LevelProgress> {
    const stats = await this.getStats();

    // Filtrer mots du niveau
    const levelWords = vocabulary.filter(w => w.jlptLevel === level);
    const totalWords = levelWords.length;

    // Compter mots maîtrisés (score >= 5)
    const masteredWords = levelWords.filter(w => {
      const score = stats.wordScores[w.hiragana] || 0;
      return score >= 5;
    }).length;

    const percentage = totalWords > 0
      ? (masteredWords / totalWords) * 100
      : 0;

    return {
      level,
      totalWords,
      masteredWords,
      percentage: Math.round(percentage * 10) / 10, // 1 décimale
      isComplete: totalWords > 0 && masteredWords === totalWords
    };
  }

  /**
   * Calcule la progression pour TOUS les niveaux JLPT
   * @returns Map<JLPTLevel, LevelProgress>
   */
  async getAllLevelsProgress(): Promise<Map<JLPTLevel, LevelProgress>> {
    const levels: JLPTLevel[] = ["kana", "n5", "n4", "n3", "n2", "n1"];
    const results = new Map<JLPTLevel, LevelProgress>();

    for (const level of levels) {
      results.set(level, await this.getLevelProgress(level));
    }

    return results;
  }
}
```

## Cas de Test

### Tests Unitaires (TDD)
```typescript
// __tests__/StatsManager.test.ts
import StatsManager from '../StatsManager';

describe('StatsManager.getLevelProgress', () => {
  let manager: StatsManager;

  beforeEach(() => {
    manager = StatsManager.getInstance();
  });

  it('retourne progression 0% pour niveau sans entraînement', async () => {
    // Setup: stats vides
    await manager.resetStats();

    const progress = await manager.getLevelProgress('kana');

    expect(progress.masteredWords).toBe(0);
    expect(progress.percentage).toBe(0);
    expect(progress.isComplete).toBe(false);
  });

  it('compte uniquement mots avec score >= 5', async () => {
    // Setup: 3 mots Kana avec scores variés
    await manager.recordAnswer('あ', true); // score 1
    await manager.recordAnswer('あ', true); // score 2
    await manager.recordAnswer('い', true); // score 1
    await manager.recordAnswer('い', true); // score 2
    await manager.recordAnswer('い', true); // score 3
    await manager.recordAnswer('い', true); // score 4
    await manager.recordAnswer('い', true); // score 5 ✓
    await manager.recordAnswer('う', true); // score 1
    // ... répéter jusqu'à う score 6 ✓

    const progress = await manager.getLevelProgress('kana');

    expect(progress.masteredWords).toBe(2); // い + う
    expect(progress.isComplete).toBe(false);
  });

  it('retourne isComplete: true si tous mots >= 5', async () => {
    // Setup: Tous les mots Kana à score 5+
    const kanaWords = vocabulary.filter(w => w.jlptLevel === 'kana');
    for (const word of kanaWords) {
      for (let i = 0; i < 5; i++) {
        await manager.recordAnswer(word.hiragana, true);
      }
    }

    const progress = await manager.getLevelProgress('kana');

    expect(progress.isComplete).toBe(true);
    expect(progress.percentage).toBe(100);
    expect(progress.masteredWords).toBe(progress.totalWords);
  });

  it('calcule progression partielle correctement', async () => {
    // Setup: 50% des mots Kana maîtrisés
    const kanaWords = vocabulary.filter(w => w.jlptLevel === 'kana');
    const halfIndex = Math.floor(kanaWords.length / 2);

    for (let i = 0; i < halfIndex; i++) {
      for (let j = 0; j < 5; j++) {
        await manager.recordAnswer(kanaWords[i].hiragana, true);
      }
    }

    const progress = await manager.getLevelProgress('kana');

    expect(progress.percentage).toBeCloseTo(50, 0);
    expect(progress.masteredWords).toBe(halfIndex);
    expect(progress.isComplete).toBe(false);
  });

  it('performance: calcul < 50ms pour niveau complet', async () => {
    const start = Date.now();
    await manager.getLevelProgress('n1'); // Plus gros niveau
    const duration = Date.now() - start;

    expect(duration).toBeLessThan(50);
  });
});

describe('StatsManager.getAllLevelsProgress', () => {
  it('retourne progression pour les 6 niveaux', async () => {
    const allProgress = await manager.getAllLevelsProgress();

    expect(allProgress.size).toBe(6);
    expect(allProgress.has('kana')).toBe(true);
    expect(allProgress.has('n1')).toBe(true);
  });
});
```
