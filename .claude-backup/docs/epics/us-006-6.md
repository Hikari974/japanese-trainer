# US-006.6: Feedback Visuel Déblocage Niveau

**Epic:** [Epic-006](./epic-006.md) - Système de Progression et Déblocage Séquentiel
**Priorité:** P2 (Nice-to-have)
**Effort Estimé:** S (2-3h)
**Statut:** DRAFT

---

## User Story

> **En tant qu'** utilisateur ayant maîtrisé un niveau
> **Je veux** recevoir un feedback visuel immédiat et gratifiant lors du déblocage d'un nouveau niveau
> **Afin de** ressentir ma progression et rester motivé

---

## Contexte Technique

### Événement Déblocage (US-006.3)
```typescript
// StatsManager.ts (existant)
interface LevelUnlockEvent {
  level: JLPTLevel;
  unlockedAt: string;
  previousLevel: JLPTLevel;
}

// Callback disponible
manager.onLevelUnlocked((event: LevelUnlockEvent) => {
  // Trigger feedback UI ici
});
```

### Feedback Souhaité
1. **Modal célébration** (full screen overlay)
2. **Animation confettis** (particules légères)
3. **Son succès** (optionnel, si volume activé)
4. **Vibration tactile** (notification success)
5. **Transition automatique** vers nouveau niveau débloqué

## Critères d'Acceptation

### Fonctionnel
- [ ] **AC1:** Modal s'affiche automatiquement lors déblocage niveau
- [ ] **AC2:** Message personnalisé "Niveau X débloqué !"
- [ ] **AC3:** Affichage nom niveau débloqué (ex: "N5 débloqué")
- [ ] **AC4:** Bouton "Commencer N5" redirige vers training avec nouveau niveau
- [ ] **AC5:** Bouton "Plus tard" ferme modal sans action

- [ ] **AC6:** Animation confettis (particules descendantes légères)
- [ ] **AC7:** Vibration tactile success (NotificationFeedbackType.Success)
- [ ] **AC8:** Son "success.mp3" joué (si volume activé)
- [ ] **AC9:** Modal peut être fermée via backdrop tap ou bouton close
- [ ] **AC10:** Animation fade-in/fade-out fluide (300ms)

- [ ] **AC11:** Composant `LevelUnlockModal.tsx` réutilisable

### Implémentation

#### Composant Modal
```typescript
// app/components/LevelUnlockModal.tsx
import React, { useEffect, useState } from 'react';
import { Modal } from 'react-native';
import { YStack, XStack, Text, Button, Dialog } from 'tamagui';
import { X, PartyPopper } from '@tamagui/lucide-icons';
import * as Haptics from 'expo-haptics';
import { Audio } from 'expo-av';
import Confetti from 'react-native-confetti';
import { JLPTLevel } from '@/core/training/types';

interface LevelUnlockModalProps {
  visible: boolean;
  level: JLPTLevel;
  previousLevel: JLPTLevel;
  onStartTraining: (level: JLPTLevel) => void;
  onDismiss: () => void;
}

export default function LevelUnlockModal({
  visible,
  level,
  previousLevel,
  onStartTraining,
  onDismiss
}: LevelUnlockModalProps) {
  const confettiRef = useRef<any>(null);

  useEffect(() => {
    if (visible) {
      // Déclencher feedback
      playSuccessSound();
      triggerHaptics();
      startConfetti();
    }
  }, [visible]);

  async function playSuccessSound() {
    try {
      const { sound } = await Audio.Sound.createAsync(
        require('@/assets/sounds/success.mp3'),
        { shouldPlay: true, volume: 0.5 }
      );

      // Unload après fin
      sound.setOnPlaybackStatusUpdate((status) => {
        if (status.isLoaded && status.didJustFinish) {
          sound.unloadAsync();
        }
      });
    } catch (error) {
      console.warn('Sound playback failed:', error);
    }
  }

  function triggerHaptics() {
    Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
  }

  function startConfetti() {
    confettiRef.current?.startConfetti();

    // Arrêter après 3s
    setTimeout(() => {
      confettiRef.current?.stopConfetti();
    }, 3000);
  }

  function handleStartTraining() {
    onStartTraining(level);
    onDismiss();
  }

  return (
    <Dialog open={visible} onOpenChange={onDismiss}>
      <Dialog.Portal>
        <Dialog.Overlay
          key="overlay"
          animation="quick"
          opacity={0.7}
          enterStyle={{ opacity: 0 }}
          exitStyle={{ opacity: 0 }}
        />

        <Dialog.Content
          key="content"
          animateOnly={['transform', 'opacity']}
          animation={[
            'quick',
            {
              opacity: {
                overshootClamping: true,
              },
            },
          ]}
          enterStyle={{ x: 0, y: -20, opacity: 0, scale: 0.9 }}
          exitStyle={{ x: 0, y: 10, opacity: 0, scale: 0.95 }}
          gap="$4"
          padding="$6"
          width="90%"
          maxWidth={400}
        >
          {/* Confettis */}
          <Confetti
            ref={confettiRef}
            count={50}
            origin={{ x: 200, y: 0 }}
            fallSpeed={3000}
            colors={['#FFD700', '#FF6B6B', '#4ECDC4', '#45B7D1']}
          />

          {/* Icône célébration */}
          <XStack justifyContent="center">
            <PartyPopper size={64} color="$yellow10" />
          </XStack>

          {/* Titre */}
          <Dialog.Title fontSize="$8" textAlign="center" fontWeight="bold">
            FÉLICITATIONS !
          </Dialog.Title>

          {/* Message */}
          <Dialog.Description fontSize="$5" textAlign="center" color="$gray11">
            Niveau {level.toUpperCase()} Débloqué !
          </Dialog.Description>

          <Text fontSize="$4" textAlign="center" color="$gray10">
            Vous avez maîtrisé tous les mots du niveau {previousLevel.toUpperCase()}.
          </Text>

          {/* Actions */}
          <YStack gap="$3" marginTop="$4">
            <Button
              size="$5"
              theme="blue"
              onPress={handleStartTraining}
              icon={<PartyPopper size={20} />}
            >
              Commencer {level.toUpperCase()}
            </Button>

            <Button
              size="$4"
              variant="outlined"
              onPress={onDismiss}
            >
              Plus tard
            </Button>
          </YStack>

          {/* Bouton close */}
          <Dialog.Close asChild>
            <Button
              position="absolute"
              top="$3"
              right="$3"
              size="$3"
              circular
              icon={<X size={20} />}
              onPress={onDismiss}
            />
          </Dialog.Close>
        </Dialog.Content>
      </Dialog.Portal>
    </Dialog>
  );
}
```

#### Hook Listener
```typescript
// app/hooks/useLevelUnlockListener.ts
import { useEffect, useState } from 'react';
import { useRouter } from 'expo-router';
import StatsManager from '@/core/training/StatsManager';
import { LevelUnlockEvent, JLPTLevel } from '@/core/training/types';

export function useLevelUnlockListener() {
  const router = useRouter();
  const [unlockEvent, setUnlockEvent] = useState<LevelUnlockEvent | null>(null);
  const [modalVisible, setModalVisible] = useState(false);

  useEffect(() => {
    const manager = StatsManager.getInstance();

    const handleUnlock = (event: LevelUnlockEvent) => {
      setUnlockEvent(event);
      setModalVisible(true);
    };

    manager.onLevelUnlocked(handleUnlock);

    // Cleanup (si StatsManager supporte removeListener)
    return () => {
      // manager.removeUnlockListener(handleUnlock);
    };
  }, []);

  function handleStartTraining(level: JLPTLevel) {
    // Naviguer vers training avec nouveau niveau
    router.push({
      pathname: '/training',
      params: { level, difficulty: 'easy' }
    });
  }

  function handleDismiss() {
    setModalVisible(false);
    setUnlockEvent(null);
  }

  return {
    unlockEvent,
    modalVisible,
    handleStartTraining,
    handleDismiss
  };
}
```

}

## Points d'Attention

### Performance
- **Son:** Unload après playback (éviter memory leak)
- **Confettis:** Arrêter après 3s (éviter surcharge)

### UX
- **Pas de Spam:** Modal une seule fois par déblocage (flag)
- **Accessibilité:** Labels ARIA pour screen readers

### Permissions
- **Son:** Check volume device (pas de son si mute)
- **Vibration:** Check permission Haptics
