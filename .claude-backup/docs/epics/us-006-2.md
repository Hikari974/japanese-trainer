### Méthodes StatsManager
```typescript
// app/core/training/StatsManager.ts
class StatsManager {
  // ... méthodes existantes ...

  /**
   * Vérifie si un niveau est débloqué
   * @param level - Niveau JLPT à vérifier
   * @returns true si niveau débloqué, false sinon
   */
  async isLevelUnlocked(level: JLPTLevel): Promise<boolean> {
    const stats = await this.getStats();
    return stats.unlockedLevels.includes(level);
  }

  /**
   * Débloque un niveau (si pas déjà débloqué)
   * @param level - Niveau à débloquer
   * @returns true si niveau nouvellement débloqué, false si déjà unlock
   */
  async unlockLevel(level: JLPTLevel): Promise<boolean> {
    const stats = await this.getStats();

    // Déjà débloqué ?
    if (stats.unlockedLevels.includes(level)) {
      return false;
    }

    // Débloquer
    stats.unlockedLevels.push(level);
    stats.levelUnlockDates[level] = new Date().toISOString();

    await this.saveStats(stats);
    return true;
  }

  /**
   * Retourne tous les niveaux débloqués (lecture seule)
   * @returns Array de JLPTLevel débloqués
   */
  async getUnlockedLevels(): Promise<JLPTLevel[]> {
    const stats = await this.getStats();
    return [...stats.unlockedLevels]; // Clone pour éviter mutation
  }

  /**
   * Migration: Initialise champs unlock si manquants
   * Appelée automatiquement lors getStats() si détection ancienne structure
   */
  private async migrateUnlockFields(stats: any): Promise<TrainingStats> {
    // Champs manquants ?
    if (!stats.unlockedLevels) {
      stats.unlockedLevels = ["kana"]; // Kana débloqué par défaut
    }

    if (!stats.levelUnlockDates) {
      stats.levelUnlockDates = {
        kana: stats.lastTrainingDate || new Date().toISOString()
      };
    }

    return stats as TrainingStats;
  }

  /**
   * Override getStats pour migration automatique
   */
  async getStats(): Promise<TrainingStats> {
    let stats = await AsyncStorage.getItem(STATS_KEY);

    if (!stats) {
      return this.getDefaultStats();
    }

    let parsed = JSON.parse(stats);

    // Migration si champs manquants
    if (!parsed.unlockedLevels || !parsed.levelUnlockDates) {
      parsed = await this.migrateUnlockFields(parsed);
      await this.saveStats(parsed); // Persist migration
    }

    return parsed;
  }

  /**
   * Stats par défaut pour nouveaux utilisateurs
   */
  private getDefaultStats(): TrainingStats {
    return {
      totalQuestions: 0,
      correctAnswers: 0,
      streakCurrent: 0,
      streakBest: 0,
      wordScores: {},
      lastTrainingDate: '',
      trainingDates: [],

      // Nouveaux champs
      unlockedLevels: ["kana"],  // Kana débloqué dès le début
      levelUnlockDates: {
        kana: new Date().toISOString()
      }
    };
  }
}
```

## Cas de Test

### Tests Unitaires
```typescript
// __tests__/StatsManager.unlock.test.ts
import StatsManager from '../StatsManager';
import AsyncStorage from '@react-native-async-storage/async-storage';

describe('StatsManager - Unlock System', () => {
  let manager: StatsManager;

  beforeEach(async () => {
    await AsyncStorage.clear();
    manager = StatsManager.getInstance();
  });

  describe('Nouveaux Utilisateurs', () => {
    it('a Kana débloqué par défaut', async () => {
      const stats = await manager.getStats();

      expect(stats.unlockedLevels).toEqual(["kana"]);
      expect(stats.levelUnlockDates.kana).toBeDefined();
    });

    it('isLevelUnlocked("kana") retourne true', async () => {
      const unlocked = await manager.isLevelUnlocked("kana");
      expect(unlocked).toBe(true);
    });

    it('isLevelUnlocked("n5") retourne false', async () => {
      const unlocked = await manager.isLevelUnlocked("n5");
      expect(unlocked).toBe(false);
    });
  });

  describe('unlockLevel', () => {
    it('débloque niveau et persiste dans AsyncStorage', async () => {
      const wasUnlocked = await manager.unlockLevel("n5");
      expect(wasUnlocked).toBe(true);

      // Vérifier persistence
      const stats = await manager.getStats();
      expect(stats.unlockedLevels).toContain("n5");
      expect(stats.levelUnlockDates.n5).toBeDefined();
    });

    it('retourne false si niveau déjà débloqué', async () => {
      await manager.unlockLevel("n5");
      const wasUnlocked = await manager.unlockLevel("n5"); // 2ème fois

      expect(wasUnlocked).toBe(false);
    });

    it('conserve ordre chronologique dans unlockedLevels', async () => {
      await manager.unlockLevel("n5");
      await manager.unlockLevel("n4");
      await manager.unlockLevel("n3");

      const levels = await manager.getUnlockedLevels();
      expect(levels).toEqual(["kana", "n5", "n4", "n3"]);
    });
  });

  describe('Migration Stats Anciennes', () => {
    it('ajoute champs unlock à stats existantes', async () => {
      // Simuler ancienne structure (sans unlock fields)
      const oldStats = {
        totalQuestions: 42,
        correctAnswers: 30,
        streakCurrent: 3,
        streakBest: 5,
        wordScores: { "あ": 8, "い": 3 },
        lastTrainingDate: "2025-11-10T12:00:00Z",
        trainingDates: ["2025-11-10"]
      };

      await AsyncStorage.setItem('@training_stats', JSON.stringify(oldStats));

      // Déclencher migration via getStats
      const stats = await manager.getStats();

      // Vérifier champs ajoutés
      expect(stats.unlockedLevels).toEqual(["kana"]);
      expect(stats.levelUnlockDates.kana).toBeDefined();

      // Vérifier données anciennes préservées
      expect(stats.totalQuestions).toBe(42);
      expect(stats.wordScores["あ"]).toBe(8);
    });

    it('ne modifie pas stats déjà migrées', async () => {
      const migratedStats = await manager.getStats(); // Déclenche migration
      const timestampBefore = migratedStats.levelUnlockDates.kana;

      // 2ème appel
      const stats2 = await manager.getStats();

      // Timestamp unlock inchangé
      expect(stats2.levelUnlockDates.kana).toBe(timestampBefore);
    });
  });

  describe('getUnlockedLevels', () => {
    it('retourne clone (pas référence directe)', async () => {
      const levels1 = await manager.getUnlockedLevels();
      levels1.push("n5"); // Mutation du retour

      const levels2 = await manager.getUnlockedLevels();
      expect(levels2).not.toContain("n5"); // Stats non mutées
    });
  });
});
```

## Points d'Attention

### Migration Safety
- **Backup:** Optionnellement sauvegarder anciennes stats dans `@training_stats_backup`
- **Idempotence:** Migration doit être safe si appelée plusieurs fois
- **Validation:** Vérifier intégrité après migration (aucun champ perdu)

### Performance
- `isLevelUnlocked` appelée fréquemment → Considérer cache in-memory
- Migration exécutée UNE SEULE fois au premier launch post-update

### TypeScript
- Utiliser `Partial<Record<JLPTLevel, string>>` pour `levelUnlockDates`
- Aucun `any` dans code migration
