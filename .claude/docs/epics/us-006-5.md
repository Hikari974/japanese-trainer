# US-006.5: Affichage Progression Détaillée par Mot

**Epic:** [Epic-006](./epic-006.md) - Système de Progression et Déblocage Séquentiel
**Priorité:** P1 (High)
**Effort Estimé:** M (4-6h)
**Statut:** DRAFT

---

## User Story

> **En tant qu'** utilisateur suivant ma progression
> **Je veux** voir le détail de maîtrise de chaque mot d'un niveau
> **Afin de** identifier quels mots nécessitent encore de l'entraînement

---

## Contexte Technique

### Donnée Disponible
```typescript
// StatsManager.ts
interface TrainingStats {
  wordScores: { [key: string]: number }; // "あ" -> 8, "い" -> 3, etc.
}

// vocabulary.json
interface VocabularyWord {
  word: string;
  hiragana: string;
  romaji: string;
  jlptLevel: JLPTLevel;
}
```

### Vue Souhaitée
- **Écran dédié:** `LevelProgressScreen.tsx` (navigation depuis `LevelSelect`)
- **Liste mots:** Tous les mots du niveau avec score actuel
- **Filtres:** Tous / Maîtrisés (>=5) / En cours (<5)
- **Tri:** Par score (croissant/décroissant) ou alphabétique
- **Performance:** Virtualisation pour gros niveaux (N1 ~2000 mots)

#### Composant Principal
```typescript
// app/components/LevelProgressView.tsx
import React, { useState, useEffect, useMemo } from 'react';
import { YStack, XStack, Text, Button, Select, ScrollView } from 'tamagui';
import { FlashList } from '@shopify/flash-list';
import StatsManager from '@/core/training/StatsManager';
import vocabulary from '@/data/vocabulary.json';
import { JLPTLevel } from '@/core/training/types';

interface LevelProgressViewProps {
  level: JLPTLevel;
}

type FilterMode = 'all' | 'mastered' | 'in-progress';
type SortMode = 'score-asc' | 'score-desc' | 'alphabetic';

interface WordProgress {
  word: string;
  hiragana: string;
  romaji: string;
  score: number;
  isMastered: boolean;
}

export default function LevelProgressView({ level }: LevelProgressViewProps) {
  const [wordScores, setWordScores] = useState<{ [key: string]: number }>({});
  const [filterMode, setFilterMode] = useState<FilterMode>('all');
  const [sortMode, setSortMode] = useState<SortMode>('score-desc');
  const [refreshing, setRefreshing] = useState(false);

  // Charger scores au mount
  useEffect(() => {
    loadWordScores();
  }, [level]);

  async function loadWordScores() {
    const manager = StatsManager.getInstance();
    const stats = await manager.getStats();
    setWordScores(stats.wordScores);
  }

  async function handleRefresh() {
    setRefreshing(true);
    await loadWordScores();
    setRefreshing(false);
  }

  // Calculer liste mots avec scores
  const wordsProgress = useMemo(() => {
    // Filtrer mots du niveau
    const levelWords = vocabulary.filter(w => w.jlptLevel === level);

    // Mapper avec scores
    let progress: WordProgress[] = levelWords.map(w => ({
      word: w.word,
      hiragana: w.hiragana,
      romaji: w.romaji,
      score: wordScores[w.hiragana] || 0,
      isMastered: (wordScores[w.hiragana] || 0) >= 5
    }));

    // Filtrer selon mode
    if (filterMode === 'mastered') {
      progress = progress.filter(w => w.isMastered);
    } else if (filterMode === 'in-progress') {
      progress = progress.filter(w => !w.isMastered);
    }

    // Trier
    if (sortMode === 'score-asc') {
      progress.sort((a, b) => a.score - b.score);
    } else if (sortMode === 'score-desc') {
      progress.sort((a, b) => b.score - a.score);
    } else {
      progress.sort((a, b) => a.hiragana.localeCompare(b.hiragana));
    }

    return progress;
  }, [level, wordScores, filterMode, sortMode]);

  // Calculer progression globale
  const totalWords = vocabulary.filter(w => w.jlptLevel === level).length;
  const masteredWords = wordsProgress.filter(w => w.isMastered).length;
  const percentage = (masteredWords / totalWords) * 100;

  return (
    <YStack flex={1} backgroundColor="$background">
      {/* Header Progression */}
      <YStack padding="$4" backgroundColor="$blue3" gap="$2">
        <Text fontSize="$8" fontWeight="bold">
          {level.toUpperCase()} - Progression Détaillée
        </Text>

        <Text fontSize="$5" color="$gray11">
          {masteredWords}/{totalWords} mots maîtrisés ({percentage.toFixed(1)}%)
        </Text>

        <Progress value={percentage} max={100} size="$3">
          <Progress.Indicator animation="quick" backgroundColor="$blue10" />
        </Progress>
      </YStack>

      {/* Filtres + Tri */}
      <XStack padding="$3" gap="$2" borderBottomWidth={1} borderColor="$gray5">
        <Button
          size="$3"
          variant={filterMode === 'all' ? 'active' : 'outlined'}
          onPress={() => setFilterMode('all')}
        >
          Tous
        </Button>
        <Button
          size="$3"
          variant={filterMode === 'mastered' ? 'active' : 'outlined'}
          onPress={() => setFilterMode('mastered')}
        >
          Maîtrisés
        </Button>
        <Button
          size="$3"
          variant={filterMode === 'in-progress' ? 'active' : 'outlined'}
          onPress={() => setFilterMode('in-progress')}
        >
          En cours
        </Button>

        <Select value={sortMode} onValueChange={setSortMode}>
          <Select.Trigger>
            <Text>Tri: {sortMode}</Text>
          </Select.Trigger>
          <Select.Content>
            <Select.Item value="score-desc">Score ↓</Select.Item>
            <Select.Item value="score-asc">Score ↑</Select.Item>
            <Select.Item value="alphabetic">Alphabétique</Select.Item>
          </Select.Content>
        </Select>
      </XStack>

      {/* Liste mots */}
      <FlashList
        data={wordsProgress}
        renderItem={({ item }) => <WordProgressItem word={item} />}
        estimatedItemSize={80}
        refreshing={refreshing}
        onRefresh={handleRefresh}
        ListEmptyComponent={
          <YStack padding="$6" alignItems="center">
            <Text fontSize="$5" color="$gray10">
              Aucun mot encore entraîné pour ce niveau
            </Text>
          </YStack>
        }
      />
    </YStack>
  );
}

// Composant item liste (mémorisé)
interface WordProgressItemProps {
  word: WordProgress;
}

const WordProgressItem = React.memo(({ word }: WordProgressItemProps) => {
  const icon = word.isMastered ? '✅' : word.score > 0 ? '🔄' : '⚪';
  const progressColor = word.isMastered ? '$green10' : '$blue10';

  return (
    <XStack
      padding="$4"
      borderBottomWidth={1}
      borderColor="$gray3"
      gap="$3"
      alignItems="center"
    >
      <Text fontSize="$8">{icon}</Text>

      <YStack flex={1} gap="$1">
        <Text fontSize="$5" fontWeight="600">
          {word.word} ({word.romaji})
        </Text>

        <XStack gap="$2" alignItems="center">
          <Text fontSize="$3" color="$gray11">
            Score: {word.score}/5
          </Text>

          <Progress
            value={Math.min(word.score / 5 * 100, 100)}
            max={100}
            size="$2"
            flex={1}
          >
            <Progress.Indicator backgroundColor={progressColor} />
          </Progress>
        </XStack>
      </YStack>
    </XStack>
  );
});
```

## Cas de Test

### Tests Composant
```typescript
// __tests__/LevelProgressView.test.tsx
import { render, fireEvent, waitFor } from '@testing-library/react-native';
import LevelProgressView from '../LevelProgressView';
import StatsManager from '../core/training/StatsManager';

jest.mock('../core/training/StatsManager');
jest.mock('../data/vocabulary.json', () => [
  { word: 'ありがとう', hiragana: 'ありがとう', romaji: 'arigatou', jlptLevel: 'n5' },
  { word: 'こんにちは', hiragana: 'こんにちは', romaji: 'konnichiwa', jlptLevel: 'n5' },
  { word: 'さようなら', hiragana: 'さようなら', romaji: 'sayounara', jlptLevel: 'n5' }
]);

describe('LevelProgressView', () => {
  beforeEach(() => {
    const mockStats = {
      wordScores: {
        'ありがとう': 8,
        'こんにちは': 3,
        'さようなら': 0
      }
    };

    StatsManager.getInstance = jest.fn().mockReturnValue({
      getStats: jest.fn().mockResolvedValue(mockStats)
    });
  });

  it('affiche progression globale niveau', async () => {
    const { getByText } = render(<LevelProgressView level="n5" />);

    await waitFor(() => {
      expect(getByText('1/3 mots maîtrisés (33.3%)')).toBeTruthy();
    });
  });

  it('affiche liste complète mots par défaut', async () => {
    const { getByText } = render(<LevelProgressView level="n5" />);

    await waitFor(() => {
      expect(getByText('ありがとう (arigatou)')).toBeTruthy();
      expect(getByText('こんにちは (konnichiwa)')).toBeTruthy();
      expect(getByText('さようなら (sayounara)')).toBeTruthy();
    });
  });

  it('filtre mots maîtrisés', async () => {
    const { getByText, queryByText } = render(<LevelProgressView level="n5" />);

    await waitFor(() => {
      fireEvent.press(getByText('Maîtrisés'));
    });

    expect(getByText('ありがとう (arigatou)')).toBeTruthy();
    expect(queryByText('こんにちは (konnichiwa)')).toBeNull();
  });

  it('trie par score décroissant', async () => {
    const { getAllByTestId } = render(<LevelProgressView level="n5" />);

    await waitFor(() => {
      const items = getAllByTestId('word-item');
      expect(items[0]).toHaveTextContent('ありがとう'); // score 8
      expect(items[1]).toHaveTextContent('こんにちは'); // score 3
      expect(items[2]).toHaveTextContent('さようなら'); // score 0
    });
  });
});
```
