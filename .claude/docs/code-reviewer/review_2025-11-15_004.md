**Issue:** Same level loading logic duplicated in two places
**Current Code:**
```tsx
// In initial useEffect (lines 36-64)
async function loadLevelStates() {
  const unlocked = await getUnlockedLevels();
  setUnlockedLevels(unlocked);

  const progressMap = new Map<Level, LevelProgress>();
  for (const level of levels) {
    const progress = await calculateProgress(level);
    progressMap.set(level, progress);

// In event listener (lines 74-87)
async function refreshLevels() {
  const unlocked = await getUnlockedLevels();
  setUnlockedLevels(unlocked);

    progressMap.set(level, progress);
```

**Recommendation:**
Extract to shared function:
```tsx

const loadLevelStates = useCallback(async () => {
  setIsLoadingLevels(true);

    const unlocked = await getUnlockedLevels();
    setUnlockedLevels(unlocked);

// Use in both places
useEffect(() => {
  loadLevelStates();
}, [loadLevelStates]);

useEffect(() => {
  const unregister = registerUnlockCallback((event) => {
    if (__DEV__) {
      console.log(`🎉 Level ${event.level} unlocked!`);
    }
    loadLevelStates(); // Reuse same function
  });

  return unregister;
}, [registerUnlockCallback, loadLevelStates]);
```

### Improvement 3: Use Stack for ProgressBar (ProgressBar.tsx)

```tsx
import React from 'react';
import { Stack, type StackProps } from 'tamagui';

export interface ProgressBarProps {
  value: number;
  height?: number;
  color?: string;
  backgroundColor?: string;
  containerProps?: StackProps;
}

export function ProgressBar({
  value,
  height = 4,
  color = '$blue10',
  backgroundColor = '$gray5',
  containerProps,
}: ProgressBarProps) {
  const clampedValue = Math.max(0, Math.min(100, value));

  return (
    <Stack
      width="100%"
      height={height}
      backgroundColor={backgroundColor}
      borderRadius={height / 2}
      overflow="hidden"
      role="progressbar"
      aria-valuemin={0}
      aria-valuemax={100}
      aria-valuenow={clampedValue}
      {...containerProps}
    >
      <Stack
        width={`${clampedValue}%`}
        height="100%"
        backgroundColor={color}
        animation="smooth"
        animateOnly={['width']}
      />
    </Stack>
  );
}
```
