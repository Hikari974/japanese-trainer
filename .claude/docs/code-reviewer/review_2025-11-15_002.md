## High Priority Issues

### 1. [PERFORMANCE] AsyncStorage appelé à chaque check de niveau débloqué

**File:** `app/services/statistics.ts:378-388`
**Issue:** La fonction `isLevelUnlocked()` charge tout le fichier AsyncStorage à chaque appel. Cette fonction est marquée comme "performance-critical" dans les commentaires et sera appelée fréquemment par l'UI (menu, écrans de sélection, etc.).

**Risk:** Performance dégradée sur appels répétés, surtout si statistics.json devient volumineux.

**Code actuel:**
```typescript
export async function isLevelUnlocked(level: JLPTLevel): Promise<boolean> {
  try {
    const stats = await loadStatistics();  // Load TOUT le fichier à chaque fois
    return stats.unlockedLevels.includes(level);
  } catch (error) {
    if (__DEV__) {
      console.error('Failed to check level unlock status:', error);
    }
    return false;
  }
}
```

**Recommandation:**

**Option 1 - Cache en mémoire (optimal pour performance):**
```typescript
// Au niveau du module
let cachedStats: UserStatistics | null = null;
let cacheTimestamp: number = 0;
const CACHE_TTL = 5000; // 5 secondes

export async function isLevelUnlocked(level: JLPTLevel): Promise<boolean> {
  try {
    const now = Date.now();

    // Utiliser cache si valide
    if (cachedStats && (now - cacheTimestamp) < CACHE_TTL) {
      return cachedStats.unlockedLevels.includes(level);
    }

    // Recharger et mettre en cache
    const stats = await loadStatistics();
    cachedStats = stats;
    cacheTimestamp = now;

    return stats.unlockedLevels.includes(level);
  } catch (error) {
    if (__DEV__) {
      console.error('Failed to check level unlock status:', error);
    }
    return false;
  }
}

// Fonction pour invalider le cache (à appeler après unlockLevel)
export function invalidateStatsCache(): void {
  cachedStats = null;
  cacheTimestamp = 0;
}
```

**Option 2 - Lecture partielle (moins optimal mais plus simple):**
```typescript
export async function isLevelUnlocked(level: JLPTLevel): Promise<boolean> {
  try {
    const stored = await AsyncStorage.getItem(STORAGE_KEY);
    if (!stored) return level === 'Kana'; // Default

    // Parse uniquement le champ nécessaire
    const parsed = JSON.parse(stored);
    const unlockedLevels = parsed.unlockedLevels ?? ['Kana'];

    return unlockedLevels.includes(level);
  } catch (error) {
    if (__DEV__) {
      console.error('Failed to check level unlock status:', error);
    }
    return false;
  }
}
```

**Impact:** Performance significativement améliorée pour les appels répétés (menu refresh, navigation, etc.)

### 2. [CODE_QUALITY] Type `any` utilisé dans migration

**File:** `app/services/statistics.ts:61`
**Issue:** Paramètre `oldStats: any` dans `migrateUnlockFields()` - TypeScript strict mode déconseille l'usage de `any`.

**Code actuel:**
```typescript
function migrateUnlockFields(oldStats: any): UserStatistics {
  const migrated = { ...oldStats };
  // ...
}
```

**Fix suggéré:**
```typescript
// Définir type pour anciennes stats (pré-unlock)
interface LegacyUserStatistics {
  words: Record<string, WordStatistic>;
  globalStats: GlobalStatistics;
  // Pas de unlockedLevels ni levelUnlockDates
}

function migrateUnlockFields(oldStats: Partial<UserStatistics>): UserStatistics {
  const migrated = { ...oldStats };

  // Add unlock fields if missing
  if (!migrated.unlockedLevels) {
    migrated.unlockedLevels = ['Kana'];
  }

  if (!migrated.levelUnlockDates) {
    migrated.levelUnlockDates = {
      Kana: oldStats.globalStats?.lastSessionDate || new Date().toISOString(),
    };
  }

  return migrated as UserStatistics;
}
```

**Impact:** Type safety améliorée, détection erreurs à la compilation.

### 3. [BEST_PRACTICES] Double appel AsyncStorage dans migration

**File:** `app/services/statistics.ts:108-113`
**Issue:** La migration appelle `saveStatistics()` immédiatement après détection, mais `loadStatistics()` est déjà dans un try/catch. Si `saveStatistics()` échoue, l'erreur est silencieuse.

**Code actuel:**
```typescript
// Migrate if unlock fields were missing
if (!parsed.unlockedLevels || !parsed.levelUnlockDates) {
  const migrated = migrateUnlockFields(parsed);
  await saveStatistics(migrated);  // Pas de gestion erreur explicite
  return migrated;
}
```

**Recommandation:**
```typescript
// Migrate if unlock fields were missing
if (!parsed.unlockedLevels || !parsed.levelUnlockDates) {
  const migrated = migrateUnlockFields(parsed);

  // Persist migration with explicit error handling
  try {
    await saveStatistics(migrated);
  } catch (saveError) {
    if (__DEV__) {
      console.warn('Migration successful but failed to persist:', saveError);
    }
    // Continue with migrated data in memory
  }

  return migrated;
}
```

**Impact:** Meilleure visibilité sur les erreurs de persistence, migration continue même si save échoue.
